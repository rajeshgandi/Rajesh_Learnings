pgbench

#create benchmark tables (-s scale factor)
/usr/pgsql-15/bin/pgbench -i -s 10 postgres

#to run benchmark
/usr/pgsql-15/bin/pgbench -c 10 -t 10 postgres

# 80 connections, run tx for 100 seconds and show progress every 3 seconds
/usr/pgsql-15/bin/pgbench -c 80 -T 100 -P 3 postgres


\watch -> run the same command every second
